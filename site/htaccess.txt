# PRJ Logistics - website settings for IONOS. Upload this file to public_html and rename it to ".htaccess" (with the dot, no .txt).

# 1. Every visitor goes to https://www.prjlogistics.co.uk (secure, with www). Other addresses (pflogistics.co.uk, no www) move permanently (301).
RewriteEngine On
RewriteCond %{HTTPS} off [OR]
RewriteCond %{HTTP_HOST} !^www\.prjlogistics\.co\.uk$ [NC]
RewriteRule ^(.*)$ https://www.prjlogistics.co.uk/$1 [R=301,L]

# 2. No folder listings: nobody can see a list of the files on the server.
Options -Indexes

# 3. Files that must never be opened from the website (scripts, backups, archives, notes).
<FilesMatch "(\.(gs|zip|bak|old|log|md|xlsx|csv|sql|env|ini)$|^(htaccess|skrypt-do-wklejenia)\.txt$|^\.)">
  Require all denied
</FilesMatch>

# 4. Security headers for every page.
<IfModule mod_headers.c>
  # browsers always use https for this site (1 year)
  Header always set Strict-Transport-Security "max-age=31536000"
  # the browser must not guess file types
  Header always set X-Content-Type-Options "nosniff"
  # other websites cannot show our pages inside a frame (protects the admin and driver login from tricks)
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Content-Security-Policy "frame-ancestors 'self'; object-src 'none'; base-uri 'self'; upgrade-insecure-requests"
  # only our own domain is sent to other sites when someone clicks a link (no reference numbers in the address)
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  # the site does not use the microphone, location or payment APIs; the camera only through the driver's photo button
  Header always set Permissions-Policy "camera=(self), microphone=(), geolocation=(), payment=(), usb=()"
  # pages are always checked for a new version, so phones never keep an old admin or driver app
  <FilesMatch "\.html$">
    Header set Cache-Control "no-cache"
  </FilesMatch>
  # private pages are never shown in Google, even if someone links to them
  <FilesMatch "^(admin|driver|client|quote|migrate)\.html$">
    Header set X-Robots-Tag "noindex, nofollow"
  </FilesMatch>
</IfModule>
